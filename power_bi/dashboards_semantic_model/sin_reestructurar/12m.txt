VAR _FechaFin =    TODAY ()
VAR _FechaInicio =    EDATE ( _FechaFin, -12 )
VAR _FechaMinContexto =    MIN ( 'dim_calendario'[Date] )
VAR _FechaMaxContexto =    MAX ( 'dim_calendario'[Date] )
VAR _Valor =
    CALCULATE (
        SUM(fact_actuacions_zones[km_inspeccionats_heli]),
        DATESINPERIOD (
            'dim_calendario'[Date],
            _FechaMaxContexto,
            -12,
            MONTH
        )
    )
RETURN
IF (
    _FechaMaxContexto < _FechaInicio
        || _FechaMinContexto > _FechaFin,
    BLANK (),
    DIVIDE ( _Valor, 1000000 )
)
