1. DELTA ANR ACUMULADO 12M - MES ANTERIOR
------------------------------------------
measure delta_anr_acum_12m_1m = ```
	
	VAR MesActualAnterior = CALCULATE(
	    SUM(fact_anr_global_mensual[anr_global_hm3]),
	    PREVIOUSMONTH(dim_calendario[Date])
	)
	VAR MesDosAnterior = 
	    CALCULATE(
	        SUM(fact_anr_global_mensual[anr_global_hm3]),
	        DATEADD(dim_calendario[Date], -2, MONTH)
	    )
	RETURN 
	    DIVIDE(MesActualAnterior - MesDosAnterior, MesDosAnterior, 0)
	```
	formatString: 0.00\ %;-0.00\ %;0.00\ %
	displayFolder: anr\acum_12m
