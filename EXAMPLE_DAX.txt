aigua_consumida_autoconsum_global_acum_12m_hm3 = 
VAR FechaMax = MAX(dim_calendario[Date])
RETURN
CALCULATE(
    SUM(fact_aigua_consum_historic[aigua_consumida_autoconsum]),
    DATESINPERIOD(
        dim_calendario[Date],
        FechaMax,
        -12,
        MONTH
    )
)/1000000  


 
aigua_consumida_autoconsum_global_acum_12m_1m_hm3= 
CALCULATE(
    [aigua_consumida_autoconsum_global_acum_12m_hm3],
    DATEADD(dim_calendario[Date], -1, MONTH)
) 
 
 



MENSUAL

aigua_consumida_autoconsum_global_mensual_hm3 = SUM(fact_aigua_consum_historic[aigua_consumida_autoconsum])/1000000    



aigua_consumida_autoconsum_global_mensual_smly_hm3 = 
    CALCULATE(
        [anr_global_mensual_smly_hm3],
        DATEADD( dim_calendario[Date], -11, MONTH )
)










SWITCH


kpi_aigua_consumida_autoconsum_global_switch = 
SWITCH(
    SELECTEDVALUE(dummy_slicer_periodo[Periodo]),
    "Mensual", [aigua_consumida_autoconsum_global_mensual_hm3],
    "12 meses", [aigua_consumida_autoconsum_global_acum_12m_hm3],
    BLANK()
)
