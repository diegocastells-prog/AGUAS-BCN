// Medidas análogas para fraude y recuperación
//
// Referencias solicitadas:
// - fact_fraudes (no fact_fraudes_all)
// - fact_vol_fraud
//
// Nota:
// - Las medidas de fact_vol_fraud usan [Data Fi Execucio Final] como fecha.
// - Se mantienen las referencias proporcionadas a [ind_resultat].
//   Verificar que esa columna exista en el modelo final de fact_vol_fraud.

// =========================================================
// Medidas base
// =========================================================

measure fraus_detectats =
    CALCULATE(
        COUNTROWS(fact_fraudes),
        fact_fraudes[ind_resultat] = "S"
    )

measure fraus_recuperats =
    CALCULATE(
        COUNTROWS(fact_vol_fraud),
        fact_vol_fraud[ind_resultat] = "S"
    )

measure volum_agua_rec_fraud =
    SUM(fact_vol_fraud[Volumen Total])

measure fraus_recupera_euros =
    CALCULATE(
        SUM(fact_vol_fraud[Imp Cobrat Frau Total]),
        fact_vol_fraud[Tipologia] = "Executada",
        NOT ISBLANK(fact_vol_fraud[Imp Cobrat Frau Total])
    )

measure pct_fraudes_recupera =
    DIVIDE(
        [fraus_recuperats],
        [fraus_detectats],
        0
    )

measure fraus_detectats_12m =
    VAR _FechaFin =
        CALCULATE(
            MAX(fact_fraudes[data_creacio_original]),
            ALL(fact_fraudes)
        )
    VAR _FechaInicio = EDATE(_FechaFin, -11)
    VAR _FechaMinContexto = MIN(dim_calendario[Date])
    VAR _FechaMaxContexto = MAX(dim_calendario[Date])
    VAR _Valor =
        CALCULATE(
            [fraus_detectats],
            DATESINPERIOD(
                dim_calendario[Date],
                _FechaMaxContexto,
                -12,
                MONTH
            )
        )
    RETURN
        IF(
            _FechaMaxContexto < _FechaInicio
                || _FechaMinContexto > _FechaFin,
            BLANK(),
            _Valor
        )
